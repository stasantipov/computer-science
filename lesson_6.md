type ArrayFactory = (size: number) => number[];
type ArrayOperation = (array: number[], size: number) => void;

type BenchmarkCase = {
    label: string;
    size: number;
    warmupSize: number;
    warmupRuns: number;
    measureRuns: number;
    operation: ArrayOperation;
};

type BenchmarkDefinition = {
    label: string;
    operation: ArrayOperation;
};

const createHoleyArray: ArrayFactory = (size) => new Array(size);
const createPackedArray: ArrayFactory = (size) => new Array(size).fill(0);

function pushToEnd(array: number[], size: number) {
    for (let i = 0; i < size; i++) {
        array.push(i);
    }
}

function popFromEnd(array: number[]) {
    while (array.length) {
        array.pop();
    }
}

function unshiftToStart(array: number[], size: number) {
    for (let i = 0; i < size; i++) {
        array.unshift(i);
    }
}

function shiftFromStart(array: number[]) {
    while (array.length) {
        array.shift();
    }
}

function writeByIndex(array: number[], size: number) {
    for (let i = 0; i < size; i++) {
        array[i] = i;
    }
}

function insertIntoMiddle(array: number[], size: number) {
    const middleIndex = Math.floor(array.length / 2);

    for (let i = 0; i < size; i++) {
        array.splice(middleIndex, 0, i);
    }
}

function removeFromMiddle(array: number[], size: number) {
    for (let i = 0; i < size; i++) {
        const middleIndex = Math.floor(array.length / 2);
        array.splice(middleIndex, 1);
    }
}

function runOperation(factory: ArrayFactory, operation: ArrayOperation, size: number) {
    const array = factory(size);
    operation(array, size);
}

function measure(factory: ArrayFactory, benchmarkCase: BenchmarkCase) {
    for (let i = 0; i < benchmarkCase.warmupRuns; i++) {
        runOperation(factory, benchmarkCase.operation, benchmarkCase.warmupSize);
    }

    const durations: number[] = [];

    for (let i = 0; i < benchmarkCase.measureRuns; i++) {
        const start = performance.now();

        runOperation(factory, benchmarkCase.operation, benchmarkCase.size);

        durations.push(performance.now() - start);
    }

    const total = durations.reduce((sum, duration) => sum + duration, 0);
    const average = total / durations.length;
    const min = Math.min(...durations);
    const max = Math.max(...durations);

    return { average, min, max };
}

const benchmarkSizes = [10, 100, 1_000, 10_000, 100_000];

const benchmarkDefinitions: BenchmarkDefinition[] = [
    { label: "Добавление в конец", operation: pushToEnd },
    { label: "Удаление с конца", operation: popFromEnd },
    { label: "Добавление в начало", operation: unshiftToStart },
    { label: "Удаление с начала", operation: shiftFromStart },
    { label: "Запись по индексу", operation: writeByIndex },
    { label: "Вставка в середину", operation: insertIntoMiddle },
    { label: "Удаление из середины", operation: removeFromMiddle },
];

const benchmarkCases: BenchmarkCase[] = benchmarkDefinitions.flatMap((definition) =>
    benchmarkSizes.map((size) => ({
        label: definition.label,
        size,
        warmupSize: Math.max(1, Math.floor(size / 10)),
        warmupRuns: 10,
        measureRuns: 5,
        operation: definition.operation,
    })),
);

const arrayKinds = [
    { label: "Дырявый массив", factory: createHoleyArray },
    { label: "Плотный массив", factory: createPackedArray },
];

for (const benchmarkCase of benchmarkCases) {
    console.log(`\n${benchmarkCase.label} (${benchmarkCase.size.toLocaleString()} элементов)`);

    for (const arrayKind of arrayKinds) {
        const stats = measure(arrayKind.factory, benchmarkCase);

        console.log(
            `${arrayKind.label}: среднее=${stats.average.toFixed(3)} мс, минимум=${stats.min.toFixed(3)} мс, максимум=${stats.max.toFixed(3)} мс`,
        );
    }
}
